# Reorder q values  
meta_qs <- metadata_with_hill %>%
    pivot_longer(cols = q0:q2, names_to = "q", values_to = "value") %>%
    filter(q %in% c("q0", "q1", "q2")) %>%
    mutate(
        qs = case_when(
            q == "q0" ~ "q0=Observed",
            q == "q1" ~ "q1=Exp Shannon",
