02. Get Differential data

# Create dataframe df_fig_pair1 to log-fold change (LFC) values
df_fig_pair1_Mangrove.system <- res_pair_Mangrove.system %>%
  dplyr::filter(`diff_Mangrove.systemSPR` == 1 |
                  `diff_Mangrove.systemTL` == 1 |
                  `diff_Mangrove.systemTL_Mangrove.systemSPR` == 1 ) %>%
  dplyr::mutate(lfc1 = ifelse(`diff_Mangrove.systemSPR` == 1, 
                              round(`lfc_Mangrove.systemSPR`, 2), NA),
                lfc2 = ifelse(`diff_Mangrove.systemTL` == 1, 
                              round(`lfc_Mangrove.systemTL`, 2), NA),
                lfc3 = ifelse(`diff_Mangrove.systemTL_Mangrove.systemSPR` == 1, 
                              round(`lfc_Mangrove.systemTL_Mangrove.systemSPR`, 2), NA),
  ) %>% tidyr::pivot_longer(cols = lfc1:lfc3, names_to = "group", 
                            values_to = "value") %>% dplyr::arrange(taxon)
# Create dataframe df_fig_pair2 to significant values
df_fig_pair2_Mangrove.system <- res_pair_Mangrove.system %>%
  dplyr::filter(`diff_Mangrove.systemSPR` == 1 |
                  `diff_Mangrove.systemTL` == 1 |
                  `diff_Mangrove.systemTL_Mangrove.systemSPR` == 1 ) %>%
  dplyr::mutate(lfc1 = ifelse(`passed_ss_Mangrove.systemSPR` == 1 & 
                                `diff_Mangrove.systemSPR` == 1, "darkblue", "black"),
                lfc2 = ifelse(`passed_ss_Mangrove.systemTL` == 1 & 
                                `diff_Mangrove.systemTL` == 1, "darkblue", "black"),
                lfc3 = ifelse(`passed_ss_Mangrove.systemTL_Mangrove.systemSPR` == 1 &
                                `diff_Mangrove.systemTL_Mangrove.systemSPR` == 1, "darkblue", "black")) %>% 
  tidyr::pivot_longer(cols = lfc1:lfc3, names_to = "group", values_to = "color") %>% 
    dplyr::arrange(taxon)
# Combine dataframes df_fig_pair1 and df_fig_pair2
df_fig_pair_Mangrove.system <- df_fig_pair1_Mangrove.system %>%
  dplyr::left_join(df_fig_pair2_Mangrove.system, by = c("taxon", "group"))

# Rename groups
df_fig_pair_Mangrove.system$group <- recode(df_fig_pair_Mangrove.system$group,
                                            `lfc1` = "SPR vs FL",
                                            `lfc2` = "TL vs FL",
                                            `lfc3` = "TL vs SPR")

df_fig_pair_Mangrove.system$group <- factor(df_fig_pair_Mangrove.system$group,
                                            levels = c("SPR vs FL",
                                                       "TL vs FL",
                                                       "TL vs SPR"))

# Filter comparisons without significant values
df_fig_pair_Mangrove.system <- df_fig_pair_Mangrove.system %>%
  group_by(group) %>%
  filter(any(!is.na(value))) %>%
  ungroup()

# Calculate limits
lo <- floor(min(df_fig_pair_Mangrove.system$value, na.rm = TRUE))
up <- ceiling(max(df_fig_pair_Mangrove.system$value, na.rm = TRUE))
mid <- (lo + up) / 2
# heatmap
heatmap_fig_pair_Mangrove.system <- df_fig_pair_Mangrove.system %>%
  ggplot(aes(x = group, y = taxon, fill = value)) +
  geom_tile(color = "gray50") +
  scale_fill_gradient2(low = "#37485FFF", high = "#907c9BFf", mid = "#cad8DCFF", #a71a50
                       na.value = "white", midpoint = mid, limit = c(lo, up),
                       name = "LFC") +
  geom_text(aes(label = round(value, 2), color = color), size = 3.5, na.rm = TRUE) +
  scale_color_identity(guide = FALSE) +
  labs(x = NULL, y = NULL, title = "Differential abundance between mangrove system") +
  theme_minimal() +
  theme(plot.title = element_text(hjust = 0.5, size = 13),
        axis.text.y = element_text(size = 12, color = "black"),
        axis.text.x = element_text(size = 9, angle = 0, 
                                   hjust = 0.5, color = "black"))

# show
print(heatmap_fig_pair_Mangrove.system)
